// Function

 create function find_class(marks int)
 returns varchar(30)
 deterministic
 begin
 if marks <= 1500 and marks >= 990 then
 return "Distinction";
 elseif marks < 990 and marks >= 900 then
 return "First Class";
 elseif marks < 900 and marks >= 825 then
 return "Higher Second Class";
 else
 return "Passed";
 end if;
 end $$

// Procedure

 create procedure get_result(IN roll_no int, OUT class varchar(20))
 begin
 declare s_name varchar(30);
 declare v_marks int;
   declare exit handler for not found
    begin
     signal sqlstate "45000" set message_text = "No Student Found";
    end;
   select stdname,marks INTO s_name,v_marks
   from stud_marks
   where stud_marks.roll_no = roll_no;
   SELECT v_marks, s_name;
   select find_class(v_marks) INTO class;
   if exists(select 1 from result where result.roll_no = roll_no) then
       update result
       set
        result.class = class, result.stdname = s_name
       where
         result.roll_no = roll_no;
   else
       insert into result(roll_no,stdname,class) values (roll_no,s_name,class);
   end if;
end $$
