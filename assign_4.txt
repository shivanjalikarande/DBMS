create procedure book_return(IN r int, IN bname varchar(30), OUT amt int)
 begin
 declare x int;
 declare fine_amt int default 0;
 declare continue handler for NOT FOUND
 begin
 signal sqlstate "45000" set message_text="No record found";
 end;
 select DATEDIFF(CURDATE(),doi) into x
 from student
 where student.roll_no = r and student.book_name = bname;
 if x < 15 then
  set fine_amt = 0;
 end if;
 if x >= 15 and x <= 30 then
   set fine_amt = (x-14) * 5;
 end if;
 if x > 30 then
   set fine_amt = (x-30) * 50 + 5*15;
 end if;
 update student
   set status = 'R'
 where roll_no = r and book_name = bname;
 insert into fine(roll_no,fine) values (r,fine_amt);
 set amt = fine_amt;
 end $$
