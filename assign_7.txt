// Triggers

/* - Database Trigger (All Types: Row level and Statement level triggers, Before and After 
Triggers). Write a database trigger on Library table. The System should keep track of the records that 
are being updated or deleted. The old value of updated or deleted records should be added in 
Library_Audit table. 
*/

----------------------------------------------------------------------------------------------------------------------------------

 CREATE TABLE Books(
 book_id int primary key auto_increment,
 book_name varchar(30) not null,
 author varchar(30));


CREATE TABLE Library_Audit(
id int primary key auto_increment,
book_id int not null,
book_name varchar(30) not null,
author varchar(30),
action_type varchar(20),
action_time timestamp default current_timestamp);

//1. Before INSERT

delimiter $$
create trigger before_insert
before insert
on Books
For Each Row
begin
if new.book_id < 0 then
signal sqlstate '45000' set message_text = "Negative ID not valid";
end if;
if new.book_name is NULL or new.book_name="" then
signal sqlstate '45000' set message_text="Please enter book name.Can't be NULL";
end if;
end$$

//2. After INSERT

create trigger after_insert
after insert
on Books
For Each Row
begin
insert into Library_Audit(book_id,book_name,author,action_type) values (new.book_id,new.book_name,new.author,"Insert");
end $$

3. Before UPDATE

create trigger before_update
before update
on Books
for each row
begin
if new.book_id < 0 then
signal sqlstate '45000' set message_text="Invalid negative book id";
end if;
if new.book_name is NULL or new.book_name="" then
signal sqlstate '45000' set message_text="Book name can't be null or empty";
end if;
end $$

----------------------------------------------------------------------------

4. After UPDATE

create trigger after_update
after update
on Books
for each row
begin
update Library_Audit set book_id = new.book_id, book_name = new.book_name,author =
new.author, action_type = "Update" where book_id = old.book_id;
end $$
