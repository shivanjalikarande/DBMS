create procedure merge_data()
   BEGIN
    declare done int default 0;
    declare o_empid int;
    declare o_fname varchar(30);
    declare o_lname varchar(30);
    declare o_salary int;
    DECLARE emp_cur CURSOR for
    select emp_id,fname,lname,salary from old_emp;
    declare continue handler for NOT FOUND set done = 1;
    OPEN emp_cur;
       cur_loop: Loop
       if done = 1 then
       leave cur_loop;
       end if;
       fetch emp_cur into o_empid,o_fname,o_lname,o_salary;
       if not exists(select emp_id from new_emp where fname = o_fname and lname = o_lname and salary = o_salary)
       then
       insert into new_emp values(o_empid,o_fname,o_lname,o_salary);
       end if;
       end Loop;
    close emp_cur;
  END //
